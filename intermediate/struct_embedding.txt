package main

import "fmt"

type person struct{
	age int
	name string
}
type Employee struct{
	employeeInfo person //Embedded struct names field
	person //Embedded struct anonymous field
	empId string
	salary float64
}
func(p person) introduce(){
	fmt.Printf("hi, Im %s and my age is %d",p.name,p.age)
}

func main() {
	emp:=Employee{
		person: person{name:"John",age: 30},
		empId: "E001",salary: 5000,
	}

	fmt.Println("Name:",emp.name)
	fmt.Println("Age:",emp.age)
	fmt.Println("Empid:",emp.empId)
	fmt.Println("salary:",emp.salary)
	emp.introduce()
}
